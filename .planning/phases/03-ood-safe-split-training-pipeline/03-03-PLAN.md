# swift sft (LoRA, bf16, vision frozen) — env-driven pixel cap:
MAX_PIXELS=$((768*768))              # == frame.engine.config.POLICY.max_pixels (parity!) — env var, NOT a CLI flag
IMAGE_MAX_TOKEN_NUM=1024             # Qwen3-VL visual-token cap (set consistently with MAX_PIXELS)
swift sft --model Qwen/Qwen3-VL-8B-Instruct --train_type lora \   # 4.4.0: --train_type (fallback --tuner_type) — VERIFY via --help
  --dataset experiments/runs/<id>/train.jsonl --val_dataset experiments/runs/<id>/id_val.jsonl \
  --torch_dtype bfloat16 --freeze_vit true --freeze_aligner true \
  --lora_rank 16 --lora_alpha 32 --lora_dropout 0.05 --target_modules all-linear \
  --learning_rate 1e-4 --lr_scheduler_type cosine --warmup_ratio 0.03 --num_train_epochs 2 \
  --per_device_train_batch_size 2 --gradient_accumulation_steps 16 --gradient_checkpointing true \
  --attn_impl flash_attn --seed 42 --output_dir experiments/runs/<id>/ckpt

# swift export (merge adapter -> standalone weights):

  --output_dir experiments/runs/<id>/merged
